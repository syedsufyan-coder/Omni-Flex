--  DDL for Trigger TRG_VALIDATE_TEACHER
--------------------------------------------------------

  CREATE OR REPLACE EDITIONABLE TRIGGER "OMNI_FLEX"."TRG_VALIDATE_TEACHER" 
BEFORE INSERT OR UPDATE ON ATTENDANCE
FOR EACH ROW
DECLARE
    v_actual_teacher VARCHAR2(10);
BEGIN
    -- Look up the teacher assigned to this enrollment's offering
    SELECT SO.TEACHER_ID 
    INTO v_actual_teacher
    FROM ENROLLMENTS E
    JOIN SECTION_OFFERINGS SO ON E.OFFERING_ID = SO.OFFERING_ID
    WHERE E.ENROLL_ID = :NEW.ENROLL_ID;

    -- If the person marking it isn't the assigned teacher, throw an error
    IF :NEW.MARKED_BY <> v_actual_teacher THEN
        RAISE_APPLICATION_ERROR(-20001, 'Unauthorized: Only the assigned teacher can mark attendance for this section.');
    END IF;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20002, 'Invalid Enrollment ID provided.');
END;

/
