#include <pthread.h>
#include <semaphore.h>
#include <stdio.h>
#include <time.h>
#include <errno.h>

// Shared Global Resources
int counter = 0;
pthread_mutex_t lock;
sem_t binary_sem;

/* * 1. STANDARD MUTEX LOCK 
 * Behavior: Blocking. If the lock is held by another thread, this thread 
 * will sleep indefinitely until the lock is released.
 */
void* standard_mutex_example(void* arg) {
    // Acquire lock (Entry Section)
    pthread_mutex_lock(&lock);
    
    // Critical Section: Atomicity is preserved here
    counter++; 
    
    // Release lock (Exit Section)
    pthread_mutex_unlock(&lock);
    return NULL;
}

/* * 2. MUTEX TRYLOCK
 * Behavior: Non-blocking. Immediately returns an error code (EBUSY) 
 * if the lock is not available. Useful to avoid "Hold and Wait".
 */
void* trylock_example(void* arg) {
    if (pthread_mutex_trylock(&lock) == 0) {
        // Successfully acquired the lock
        counter++;
        pthread_mutex_unlock(&lock);
        printf("TryLock: Critical Section Accessed.\n");
    } else {
        // Lock was busy, thread performs alternative work
        printf("TryLock: Lock Busy. Performing alternative task...\n");
    }
    return NULL;
}

/* * 3. MUTEX TIMEDLOCK
 * Behavior: Conditional Blocking. Waits for the lock for a specific 
 * duration before timing out. Prevents indefinite Deadlock.
 */
void* timedlock_example(void* arg) {
    struct timespec ts;
    
    // Get current time and add a 5-second timeout
    clock_gettime(CLOCK_REALTIME, &ts);
    ts.tv_sec += 5;

    int result = pthread_mutex_timedlock(&lock, &ts);

    if (result == 0) {
        counter++;
        pthread_mutex_unlock(&lock);
        printf("TimedLock: Success within 5 seconds.\n");
    } else if (result == ETIMEDOUT) {
        printf("TimedLock: Failed. The 5-second wait expired.\n");
    }
    return NULL;
}

/* * 4. SEMAPHORE (BINARY/COUNTING)
 * Behavior: Signaling mechanism. 'sem_wait' decrements the value, 
 * 'sem_post' increments it.
 */
void* semaphore_example(void* arg) {
    // wait() operation: If value > 0, decrement and enter. Else, block.
    sem_wait(&binary_sem);
    
    // --- Critical Section ---
    counter++;
    
    // signal() operation: Increment value and wake up a waiting thread.
    sem_post(&binary_sem);
    return NULL;
}

    // Initialization
    pthread_mutex_init(&lock, NULL);
    
    // Initialize Semaphore: (Pointer, pshared, initial_value)
    // If initial_value = 1, it acts like a Mutex.
    sem_init(&binary_sem, 0, 1);

    pthread_mutex_destroy(&lock);
